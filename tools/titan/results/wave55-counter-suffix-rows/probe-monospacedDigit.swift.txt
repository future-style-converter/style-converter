    @MainActor
    private func ink<V: View>(_ v: V) throws -> (Int, Int) {
        let view = v.padding(8).background(Color.white)
        let r = ImageRenderer(content: view); r.scale = 1
        let cg = try XCTUnwrap(r.cgImage)
        var buf = [UInt8](repeating: 0, count: cg.width * cg.height * 4)
        let ctx = try XCTUnwrap(CGContext(data: &buf, width: cg.width, height: cg.height, bitsPerComponent: 8, bytesPerRow: cg.width * 4, space: CGColorSpaceCreateDeviceRGB(), bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue))
        ctx.draw(cg, in: CGRect(x: 0, y: 0, width: cg.width, height: cg.height))
        var lo = Int.max, hi = -1
        for y in 0..<cg.height { for x in 0..<cg.width { let i = (y*cg.width+x)*4; if buf[i] < 128 && buf[i+1] < 128 && buf[i+2] < 128 { lo = min(lo,x); hi = max(hi,x) } } }
        return (lo, hi)
    }

        {"id":"run","name":"run","text":"\(text)",
         "properties":[{"type":"WhiteSpace","data":"PRE"},
           {"type":"LineHeight","data":{"original":{"type":"length","px":20}}},
           {"type":"Color","data":{"srgb":{"r":0,"g":0,"b":0},"original":{"r":0,"g":0,"b":0}}},
           {"type":"FontFamily","data":\(family)},
           {"type":"FontSize","data":{"px":16,"original":{"type":"length","px":16}}},
           {"type":"FontStyle","data":"normal"},{"type":"FontWeight","data":400}\(fvn)]}
        """.utf8))
    }

    /// (box height, ink rows, ink cols) of a marker Text exactly as markerPlacement builds it.
    @MainActor
    private func markerBox(_ str: String, size: CGFloat, rtl: Bool) throws -> String {
        let t = Text(ScriptFallbackFonts.annotate(str, enabled: true, size: size))
            .font(.custom("Inter-Regular", size: size))
            .foregroundColor(.black)
            .fixedSize(horizontal: true, vertical: true)
            .background(Color(red: 1, green: 0, blue: 0))
            .environment(\.layoutDirection, rtl ? .rightToLeft : .leftToRight)
        let r = ImageRenderer(content: t.padding(8).background(Color.white)); r.scale = 1
        let cg = try XCTUnwrap(r.cgImage)
        var buf = [UInt8](repeating: 0, count: cg.width * cg.height * 4)
        let ctx = try XCTUnwrap(CGContext(data: &buf, width: cg.width, height: cg.height, bitsPerComponent: 8, bytesPerRow: cg.width * 4, space: CGColorSpaceCreateDeviceRGB(), bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue))
        ctx.draw(cg, in: CGRect(x: 0, y: 0, width: cg.width, height: cg.height))
        var bTop = Int.max, bBot = -1, bL = Int.max, bR = -1, iTop = Int.max, iBot = -1, iL = Int.max, iR = -1
        for y in 0..<cg.height { for x in 0..<cg.width { let i = (y*cg.width+x)*4
            let rr = buf[i], gg = buf[i+1], bb = buf[i+2]
            if !(rr > 240 && gg > 240 && bb > 240) { bTop = min(bTop,y); bBot = max(bBot,y); bL = min(bL,x); bR = max(bR,x) }
            if rr < 128 && gg < 128 && bb < 128 { iTop = min(iTop,y); iBot = max(iBot,y); iL = min(iL,x); iR = max(iR,x) } } }
        return "box \(bR-bL+1)x\(bBot-bTop+1) ink y\(iTop-bTop)-\(iBot-bTop) x\(iL-bL)-\(iR-bL)"
    }

    @MainActor
    func testProbeR1R2() throws {
        var dir = URL(fileURLWithPath: #filePath)
        for _ in 0..<5 { dir.deleteLastPathComponent() }
        for f in ["Inter-Regular", "Inter-Medium", "Inter-Bold", "Inter-Black"] {
            let url = dir.appendingPathComponent("apps/ios-harness/StyleConverterTest/Resources/\(f).otf")
            CTFontManagerRegisterFontsForURL(url as CFURL, .process, nil)
        }
        for (s, rtl) in [("א.", false), ("ב.", false), ("•", true), ("一、", false)] {
            let p = ListMarkerBidi.painted(s, itemIsRTL: rtl)
            print("PROBE R1R2 \(s) rtl=\(rtl) raw:", try markerBox(s, size: 16, rtl: rtl), "| painted:", try markerBox(p, size: 16, rtl: rtl), "| changed:", p != s)
        }
    }
