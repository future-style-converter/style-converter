    @MainActor
    private func raster<V: View>(_ v: V) throws -> (px: [UInt8], w: Int, h: Int) {
        let r = ImageRenderer(content: v); r.scale = 1
        let cg = try XCTUnwrap(r.cgImage)
        let w = cg.width, h = cg.height
        var buf = [UInt8](repeating: 0, count: w*h*4)
        let ctx = try XCTUnwrap(CGContext(data: &buf, width: w, height: h, bitsPerComponent: 8,
                                          bytesPerRow: w*4, space: CGColorSpaceCreateDeviceRGB(),
                                          bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue))
        ctx.draw(cg, in: CGRect(x: 0, y: 0, width: w, height: h))
        return (buf, w, h)
    }

    private func sha(_ b: [UInt8]) -> String {
        SHA256.hash(data: Data(b)).map { String(format: "%02x", $0) }.joined()
    }

    /// bbox of pixels matching `near` within tol; nil if none.
    private func box(_ img: (px: [UInt8], w: Int, h: Int), near: (Int, Int, Int), tol: Int)
        -> (w: Int, h: Int, minX: Int, minY: Int, count: Int)? {
        var minX = Int.max, minY = Int.max, maxX = -1, maxY = -1, n = 0
        for y in 0..<img.h { for x in 0..<img.w {
            let i = (y*img.w + x)*4
            let r = Int(img.px[i]), g = Int(img.px[i+1]), b = Int(img.px[i+2])
            guard abs(r-near.0) < tol, abs(g-near.1) < tol, abs(b-near.2) < tol else { continue }
            n += 1; minX = min(minX,x); minY = min(minY,y); maxX = max(maxX,x); maxY = max(maxY,y)
        } }
        return maxX >= 0 ? (maxX-minX+1, maxY-minY+1, minX, minY, n) : nil
    }

    /// bbox of non-white ink.
    private func inkBox(_ img: (px: [UInt8], w: Int, h: Int))
        -> (w: Int, h: Int, minX: Int, minY: Int, count: Int)? {
        var minX = Int.max, minY = Int.max, maxX = -1, maxY = -1, n = 0
        for y in 0..<img.h { for x in 0..<img.w {
            let i = (y*img.w + x)*4
            let r = Int(img.px[i]), g = Int(img.px[i+1]), b = Int(img.px[i+2])
            guard r < 250 || g < 250 || b < 250 else { continue }
            n += 1; minX = min(minX,x); minY = min(minY,y); maxX = max(maxX,x); maxY = max(maxY,y)
        } }
        return maxX >= 0 ? (maxX-minX+1, maxY-minY+1, minX, minY, n) : nil
    }

    @MainActor

        try raster(ComponentRenderer(component: c)

            .background(Color.white))
    }

    private func roots(_ path: String) throws -> [IRComponent] {
        let data = try Data(contentsOf: URL(fileURLWithPath: path))
        let doc = try JSONDecoder().decode(IRDocument.self, from: data)
        return doc.components.filter { $0.slot == nil }
    }
